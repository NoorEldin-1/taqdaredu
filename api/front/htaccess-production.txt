<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /

    # ===== Force HTTPS (redirect all HTTP traffic to HTTPS) =====
    RewriteCond %{HTTPS} !=on
    RewriteCond %{HTTP:X-Forwarded-Proto} !https
    RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

    # ===== Canonical host: www → non-www (avoid duplicate-host indexing) =====
    RewriteCond %{HTTP_HOST} ^www\.(.+)$ [NC]
    RewriteRule ^ https://%1%{REQUEST_URI} [L,R=301]

    # ===== SEO bot rewrite — must run FIRST =====
    # Bots get a single PHP entry point that internally fetches /api/seo/<route>.
    # EVERY non-file content path for a bot goes to /seo-router.php, which renders a
    # known SEO route OR returns a real 404 for unknown paths (no more soft-404 SPA).
    # We exclude: real files (!-f → assets, robots.txt, favicon, llms.txt…), the API +
    # static virtual prefixes, and the generated sitemap. Real users skip this whole
    # block and fall through to the React SPA. (Note: NO !-d check — the site root "/"
    # resolves to a directory and must still reach seo-router for the homepage.)
    RewriteCond %{HTTP_USER_AGENT} (Googlebot|Google-InspectionTool|Storebot-Google|GoogleOther|Google-Extended|Bingbot|DuckDuckBot|YandexBot|Slurp|Baiduspider|Sogou|Exabot|facebookexternalhit|Twitterbot|LinkedInBot|WhatsApp|TelegramBot|Applebot|PetalBot|Bytespider|Amazonbot|SemrushBot|AhrefsBot|MJ12bot|GPTBot|OAI-SearchBot|ChatGPT-User|ClaudeBot|Claude-Web|anthropic-ai|CCBot|PerplexityBot) [NC]
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteCond %{REQUEST_URI} !^/(api|_app|uploads|assets|img|api_courses|api_frontend|api_payment|api_notifications|api_messages|api_admin|api_reports|api_webhooks)(/|$) [NC]
    RewriteCond %{REQUEST_URI} !^/(sitemap\.xml|login/google_callback)$ [NC]
    RewriteRule ^(.*)$ /seo-router.php?path=$1 [L,QSA]

    # /sitemap.xml → fresh canonical sitemap from Sitemap_seo controller
    RewriteRule ^sitemap\.xml$ /api/index.php/sitemap_seo [L]

    # ===== Standard rules =====
    RewriteRule ^_app/ - [L]
    RewriteRule ^login/google_callback /api/login/google_callback [L,QSA]
    RewriteRule ^api/ - [L]

    # Frontend now calls /api/api_<module>/<endpoint> directly (no rewrite needed —
    # the subdirectory's own .htaccess dispatches to CodeIgniter's index.php).
    # Old /api_<module>/* URLs are 307-redirected for backward compatibility
    # with bookmarks / external callers; new code should use /api/api_<module>.
    RewriteRule ^api_courses/(.*)$ /api/api_courses/$1 [L,QSA,R=307]
    RewriteRule ^api_frontend/(.*)$ /api/api_frontend/$1 [L,QSA,R=307]
    RewriteRule ^api_payment/(.*)$ /api/api_payment/$1 [L,QSA,R=307]
    RewriteRule ^api_notifications/(.*)$ /api/api_notifications/$1 [L,QSA,R=307]
    RewriteRule ^api_messages/(.*)$ /api/api_messages/$1 [L,QSA,R=307]
    RewriteRule ^api_admin/(.*)$ /api/api_admin/$1 [L,QSA,R=307]
    RewriteRule ^api_reports/(.*)$ /api/api_reports/$1 [L,QSA,R=307]
    RewriteRule ^api_webhooks/(.*)$ /api/api_webhooks/$1 [L,QSA,R=307]

    RewriteRule ^uploads/(.*)$ /api/uploads/$1 [L]
    RewriteRule ^assets/(.*)$ /api/assets/$1 [L]

    # Clean image-proxy URL: /img/uploads/<path><ext>.jpg → img.php?src=uploads/<path><ext>
    # Ends in .jpg with no query string so strict crawlers (WhatsApp) accept og:image.
    RewriteRule ^img/(uploads/.+)\.jpg$ /img.php?src=$1 [L]

    RewriteCond %{REQUEST_FILENAME} -f [OR]
    RewriteCond %{REQUEST_FILENAME} -d
    RewriteRule ^ - [L]

    RewriteRule ^ /index.html [L]
</IfModule>
